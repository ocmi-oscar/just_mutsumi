    def _parse_and_execute_tools(text):
        """从AI回复中解析[TOOL:xxx:yyy]并执行对应操作"""
        import re
        tools = re.findall(r'\[TOOL:(\w+):([^\]]*)\]', text)
        clean_text = re.sub(r'\[TOOL:[^\]]*\]', '', text).strip()

        for tool, param in tools:
            try:
                _execute_tool(tool, param)
            except Exception as e:
                pass

        return clean_text, tools

        if not api_key:

        headers = {

        url = base_url.rstrip("/") + "/chat/completions"

        try:
            with urllib.request.urlopen(req, timeout=30, context=_ai_ssl_ctx) as r:
                resp = _json.loads(r.read().decode("utf-8"))
            raw_reply = resp["choices"][0]["message"]["content"]
        except urllib.error.HTTPError as e:
            err = e.read().decode("utf-8")[:80]
            store._ai_error    = "请求失败：" + err
            store._ai_thinking = False
            renpy.restart_interaction()
            return
        except Exception as e:
            store._ai_error    = "连接失败：" + str(e)[:60]
            store._ai_thinking = False
            renpy.restart_interaction()
            return

        # 解析工具调用
        clean_reply, tools = _parse_and_execute_tools(raw_reply)

        # 更新对话历史（本地）
        hist = list(persistent.ai_chat_history or [])
        hist.append({"role": "user",      "content": user_msg})
        hist.append({"role": "assistant", "content": clean_reply})
        # 只保留最近50条
        persistent.ai_chat_history = hist[-50:]

        # 好感度奖励：每次成功对话+0.5，每天最多10点
        import datetime as _dt_ai
        _today_str = str(_dt_ai.date.today())
        _ai_gw_key = "ai_gw_date"
        _ai_gw_count_key = "ai_gw_count"
        _last_date = getattr(persistent, _ai_gw_key, "")
        _today_count = getattr(persistent, _ai_gw_count_key, 0) if _last_date == _today_str else 0
        if _today_count < 10:
            _gw_add = min(0.5, 10 - _today_count)
            if hasattr(persistent, 'gw_wakaba'):
                persistent.gw_wakaba = float(persistent.gw_wakaba or 0) + _gw_add
            setattr(persistent, _ai_gw_key, _today_str)
            setattr(persistent, _ai_gw_count_key, _today_count + _gw_add)

        renpy.save_persistent()

        store._ai_last_reply = clean_reply
        store._ai_thinking   = False

        # 如果有工具触发，加一条提示

            tool_names = {"white_noise": "白噪音已开启", "stop_noise": "白噪音已关闭",
                          "pomodoro": "番茄钟已启动", "wallpaper": "壁纸已切换",
                          "diary": "日记已打开"}
            notices = [tool_names.get(t, t) for t, _ in tools]
            store._ai_tool_notice = "  ·  ".join(notices)
        else:
            store._ai_tool_notice = ""

        renpy.restart_interaction()

                _raw = store._ai_reply_raw

                # 解析 [GW:+N] 或 [GW:-N] 指令
                _gw_matches = _re2.findall(r'\[GW:([+-]?\d+(?:\.\d+)?)\]', _raw)

                _clean = _re2.sub(r'\[GW:[^\]]+\]', '', _raw).strip()

    def _ai_send_from_screen():
        msg = (store._ai_input_buf or "").strip()
        if not msg:
            return
        _safe_msg = msg.replace("{","｛").replace("}","｝").replace("[","［").replace("]","］")
        if not isinstance(store._ai_chat_log, list):
            store._ai_chat_log = []
        store._ai_chat_log.append(("user", _safe_msg))
        store._ai_chat_log = store._ai_chat_log[-20:]
        store._ai_input_buf  = ""
        store._ai_thinking   = True
        store._ai_reply_done = False
        store._ai_reply_raw  = ""

        import threading as _ai_th4, re as _re4
        def _ai_do4():
            try:
                _do_ai_chat(msg)
                raw = store._ai_last_reply or ""
            except:
                raw = ""
            gw_matches = _re4.findall(r'\[GW:([+-]?\d+(?:\.\d+)?)\]', raw)
            for gw_m in gw_matches:
                try:
                    delta = float(gw_m)
                    cur = float(getattr(persistent, 'gw_wakaba', 0) or 0)
                    persistent.gw_wakaba = max(0, cur + delta)
                    renpy.save_persistent()
                except:
                    pass
            clean = _re4.sub(r'\[GW:[^\]]+\]', '', raw)
            clean = _re4.sub(r'\[TOOL:[^\]]+\]', '', clean).strip()
            clean = clean.replace("{","｛").replace("}","｝").replace("[","［").replace("]","］")
            if clean:
                store._ai_chat_log.append(("mutsumi", clean))
                store._ai_chat_log = store._ai_chat_log[-20:]
            store._ai_display_reply = clean
            store._ai_reply_done    = True
            store._ai_thinking      = False
            renpy.restart_interaction()
        t4 = _ai_th4.Thread(target=_ai_do4)
        t4.daemon = True
        t4.start()
